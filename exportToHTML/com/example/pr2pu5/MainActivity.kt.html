class Preferencias(context: Context) {
    private val preferences = context.getSharedPreferences("app", Context.MODE_PRIVATE)
    fun guardar(prop: String, value: String) {
        preferences.edit()
            .putString(prop, value)
            .apply()
    }
    fun obtener(prop: String, def: String): String? {
        return preferences.getString(prop, def)
    }


    fun remover(prop: String) {
        preferences.edit()
            .remove(prop)
            .apply()
    }
}

@Composable
fun AppContent(modifier: Modifier = Modifier) {
    val context = LocalContext.current //val es un constante
    val preferencias = Preferencias(context)
    var destination = "LoginContent" //var es un variable


    if (preferencias.obtener("id", "") != "") { //if an id already exists in a session
        destination = "MenuContent"                         //takes you straight to you Menu
    }



    val navController = rememberNavController()

    NavHost(
        navController = navController,
        startDestination = destination,
        modifier = modifier
    ) {
        composable("LoginContent") { LoginContent(navController) }
        composable("MenuContent") { MenuContent(navController) }
        composable("PacienteContent") { PacienteContent(navController) }
        composable("PacienteFormContent") { PacienteFormContent(navController) }
        composable("HistorialContent") { HistorialContent(navController) }
        composable("HistorialFormContent") { HistorialFormContent(navController) }
        composable("HabitacionContent") { HabitacionContent(navController) }
        composable("HabitacionFormContent") { HabitacionFormContent(navController) }
    }
}




interface ApiService {
    @POST("iniciarSesion.php")
    @FormUrlEncoded
    suspend fun iniciarSesion(
        @Field("nombreUsuario") nombreUsuario: String,
        @Field("contrasena") contrasena: String
    ): Response<String>


    @GET("categorias.php?buscar&busqueda")
    suspend fun buscarCategorias(): List<ModeloCategoria>


    @POST("productos.php?registrar")
    @FormUrlEncoded
    suspend fun registrarProducto(
        @Field("nombre") nombre: String,
        @Field("precio") precio: Double,
        @Field("categoria") categoria: Int
    ): Response<Unit>


    @GET("productos.php?buscar&busqueda")
    suspend fun buscarProductos(): List<ModeloProducto>


    @POST("productos.php?modificar")
    @FormUrlEncoded
    suspend fun modificarProducto(
        @Field("id") id: Int,
        @Field("nombre") nombre: String,
        @Field("precio") precio: Double,
        @Field("categoria") categoria: Int
    ): Response<Unit>


    @POST("productos.php?eliminar")
    @FormUrlEncoded
    suspend fun eliminarProducto(
        @Field("id") id: Int
    ): Response<Unit>
}

@Composable
fun LoginContent(navController: NavHostController, modifier: Modifier = Modifier) {
    val context = LocalContext.current
    val preferencias = Preferencias(context)
    val scope = rememberCoroutineScope ()

    var usuario by remember { mutableStateOf("") }
    var contrasena by remember { mutableStateOf("") }

    Box(
        modifier = modifier.fillMaxSize()
    ) {
        Image(
            painter = painterResource(id = R.drawable.perfil),
            contentDescription = "Fondo de inicio de sesión",
            modifier = Modifier.fillMaxSize(),
            contentScale = ContentScale.Crop
        )

        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(24.dp),
            horizontalAlignment = Alignment.Start,
            verticalArrangement = Arrangement.Center
        ) {
            Text(
                text = "Inicio de Sesión",
                fontSize = 24.sp,
                fontWeight = FontWeight.ExtraBold,
                color = Color.White,
                modifier = Modifier
                    .align(Alignment.CenterHorizontally)
                    .padding(bottom = 24.dp)
            )

            Text(
                text = "Usuario:",
                color = Color.White,
                modifier = Modifier.padding(top = 16.dp)
            )
            TextField(
                value = usuario,
                onValueChange = { usuario = it },
                placeholder = { Text("Ingresa tu Usuario") },
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(bottom = 8.dp)
            )

            Text(
                text = "Contraseña:",
                color = Color.White
            )
            TextField(
                value = contrasena,
                onValueChange = { contrasena = it },
                placeholder = { Text("Contraseña") },
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(bottom = 16.dp),
                visualTransformation = PasswordVisualTransformation()
            )

            Button(
                onClick = {



                scope.launch {
                    try {
                        val respuesta : Response<String> = api.iniciarSesion(usuario, contrasena)
                        val id : String = respuesta.body().toString()

                        if (id.toDoubleOrNull() != null) {
                            preferencias.guardar("id", id)
                            navController.navigate("MenuContent")
                        }
                        else {
                            Toast.makeText(context, "Usuario y/o contraseña incorrecto(s).", Toast.LENGTH_SHORT).show()
                            }
                        }
                        catch (e: Exception) {
                            Log.e("API", "Error al iniciar sesión: ${e.message}")
                        }
                }

                //




                },
                modifier = Modifier.align(Alignment.End)
            ) {
                Text("Iniciar sesión")
            }
        }
    }
}

        TextField(

            onValueChange = { fechaIngreso = it },
            placeholder = { Text("Ej. 01/10/2026") },
            modifier = Modifier
                .fillMaxWidth()
                .padding(bottom = 8.dp)
        )

        Text(text = "Fecha de Salida:")
        TextField(
            value = fechaSalida,
            onValueChange = { fechaSalida = it },
            placeholder = { Text("Ej. 05/10/2026") },
            modifier = Modifier
                .fillMaxWidth()
                .padding(bottom = 8.dp)
        )

        Text(text = "Nombre:")
        TextField(
            value = nombre,
            onValueChange = { nombre = it },
            placeholder = { Text("Ej. Juan Pérez") },
            modifier = Modifier
                .fillMaxWidth()
                .padding(bottom = 8.dp)
        )

        Text(text = "Domicilio:")
        TextField(
            value = domicilio,
            onValueChange = { domicilio = it },
            placeholder = { Text("Ej. Calle 1") },
            modifier = Modifier
                .fillMaxWidth()
                .padding(bottom = 16.dp)
        )

        Button(
            onClick = {
                if  (curp.isNotBlank() && nombre.isNotBlank()) {
                    listaPacienteGlobal.add(
                        PacienteItem(
                            curp = curp,
                            fechaIngreso = fechaIngreso,
                            fechaSalida = fechaSalida,
                            nombre = nombre,
                            domicilio = domicilio
                        )
                    )
                    Toast.makeText(context, "Paciente guardado con éxito", Toast.LENGTH_SHORT)
                        .show()
                    navController.popBackStack()
                } else {
                    Toast.makeText(context, "Completa los campos obligatorios", Toast.LENGTH_SHORT).show()
                }
            },
            modifier = Modifier.fillMaxWidth()
        ) {
            Text("Guardar Paciente")
        }

        Spacer(modifier = Modifier.height(8.dp))

        Button(
            onClick = { navController.popBackStack() },
            colors = ButtonDefaults.buttonColors(containerColor = Color.Gray),
            modifier = Modifier.fillMaxWidth()
        ) {
            Text("Cancelar")
        }
    }
}
